
============================================================
 CANDIDATE 1
============================================================

* 360V to 24V SEPIC, 500W
Vin in 0 360
L1 in sw1 220u
M1 sw1 gate1 0 0 NMOS W=1 L=1
C1 sw1 sw2 22u
L2 sw2 0 220u
M2 sw2 gate2 0 0 NMOS W=1 L=1
D1 sw2 out DIODE
C2 out 0 100u
Rload out 0 1.15
Rbleed sw1 0 1Meg
Rbleed1 sw2 0 1Meg
Vgate  gate1 0 PULSE(0 12 0 1n 1n 5.8u 10u)
Vgate1 gate2 0 PULSE(0 12 0 1n 1n 5.8u 10u)
.model NMOS NMOS(Vto=1 Kp=2 Lambda=0)
.model DIODE D
.tran 10n 1m
.end
