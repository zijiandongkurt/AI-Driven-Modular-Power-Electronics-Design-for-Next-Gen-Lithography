* 250V to 3.3V interleaved 2-phase buck, 200W
Vin in 0 250
M1 in gate1 sw1 0 NMOS W=1 L=1
M2 in gate2 sw2 0 NMOS W=1 L=1
D1 0 sw1 DIODE
D2 0 sw2 DIODE
L1 sw1 out 22u
L2 sw2 out 22u
C1 out 0 100u
Rload out 0 0.0545
Vgate1 gate1 0 PULSE(0 12 0 1n 1n 5.8u 10u)
Vgate2 gate2 0 PULSE(0 12 5.8u 1n 1n 5.8u 10u)
.model NMOS NMOS(Vto=1 Kp=2 Lambda=0)
.model DIODE D
.tran 10n 10m
.save V(*) I(*)
.end
